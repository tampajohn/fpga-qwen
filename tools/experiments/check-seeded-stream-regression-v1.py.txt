#!/usr/bin/env python3
"""Guarded full sampled-stream regression; no absolute accuracy or timing claim."""
import argparse,json,sys,threading
from pathlib import Path
ROOT=Path(__file__).resolve().parents[2];sys.path.insert(0,str(ROOT/'tools'))
from capture_native_cpu_logits import guard,read_ids,save,sha
from check_native_cpu_prefix import Recorder,Protocol,compare,require
from chat_protocol import ChatRequest
BASE=ROOT/'build/native-cpu'
def main():
 p=argparse.ArgumentParser(description=__doc__);p.add_argument('--candidate',type=Path,required=True);p.add_argument('--baseline',type=Path,default=BASE/'build-19/qwen_cpu');p.add_argument('--directory',type=Path,required=True);a=p.parse_args()
 directory=a.directory.resolve();directory.mkdir(parents=True,exist_ok=False);out=directory/'report.json'
 binaries={'baseline':a.baseline.resolve(),'candidate':a.candidate.resolve()};weights=BASE/'weights.bin'
 fixtures=[('long',ROOT/'build/native-phase0/prompt-long.json'),('context2048',BASE/'extended-tutorial-fixtures/context2048.json'),('context3968',BASE/'extended-tutorial-fixtures/context3968.json')]
 report=dict(completed=False,passed=False,performance_claim_eligible=False,absolute_accuracy_qualified=False,gpu=False,pytorch=False,guard_before=guard(BASE/'../benchmark-status'),requested_tokens=20,seeds=[42,1337],repeats=2,threads=12,prefill_threads=6,cases=[],inputs={},limitations=['Known baseline absolute63/64 oracle failures remain failed.','Fixed token fixtures, not additional semantic answer-quality evidence.','EOS is respected and recorded; at least two contexts must complete20 tokens for each seed.','Prefix reuse and suffix drafting disabled here; separate lifecycle/speculation gates required.'])
 active=None
 try:
  paths=[Path(__file__).resolve(),ROOT/'tools/capture_native_cpu_logits.py',ROOT/'tools/check_native_cpu_prefix.py',ROOT/'tools/native_cpu_chat.py',ROOT/'tools/chat_protocol.py',ROOT/'tools/suffix.py',ROOT/'tools/benchmark_guard.py',weights,Path(str(weights)+'.index'),Path(str(weights)+'.manifest.json')]+[x[1] for x in fixtures]
  for binary in binaries.values():
   mp=binary.parent/'manifest.json';m=json.loads(mp.read_text());require(m['completed'] and m['binary_sha256']==sha(binary),'Invalid build binding');paths.extend([binary,mp])
   for name,digest in m['source_sha256'].items():require(sha(Path(name))==digest,'Build source changed');paths.append(Path(name))
  report['inputs']={str(p.resolve()):sha(p) for p in paths};report['binaries']={k:str(v) for k,v in binaries.items()};save(out,report)
  for label,prompt in fixtures:
   ids=read_ids(prompt);require(len(ids)+19<=4096,'Context overflow')
   for seed in report['seeds']:
    case=dict(fixture=label,prompt_ids=ids,seed=seed,arms=[],completed=False);report['cases'].append(case);save(out,report)
    reference=None
    for repeat in range(2):
     for role in ('baseline','candidate'):
      active=Recorder(Protocol(),binary=binaries[role],weights=weights,threads=12,prefill_threads=6,reuse_prefix=False)
      active.begin(directory/f'{label}-seed{seed}-{role}-repeat{repeat}')
      events=list(active.stream(ChatRequest(f'{label}-{seed}',ids,20,0,decoding='chat',seed=seed),threading.Event()))
      require(events and events[-1].get('type')=='done','Missing terminal stream event');done=events[-1]
      require(done['finish_reason'] in ('length','eos'),'Unexpected stop');require(done['metrics']['output_tokens']==len(done['token_ids']),'Token count mismatch')
      require(done['finish_reason']!='length' or len(done['token_ids'])==20,'Incomplete length stream')
      result=dict(finish_reason=done['finish_reason'],text=done['text'],token_ids=done['token_ids'],metrics=done['metrics'],predictions=list(active.records))
      require(result['predictions'],'No predictions');require(result['metrics']['reused_prefix_tokens']==0,'Fresh request reused prefix');require(active.ready.get('prefix_reuse_chunk')==512,'Unexpected canonical chunk')
      case['arms'].append(dict(role=role,repeat=repeat,startup=active.ready,result=result));save(out,report)
      active.close();active=None
      if reference is None:reference=result
      else:compare(reference,result)
    case.update(completed=True,passed=True,full_twenty_tokens=len(reference['token_ids'])==20);save(out,report)
  for seed in report['seeds']:require(sum(c['full_twenty_tokens'] for c in report['cases'] if c['seed']==seed)>=2,'Insufficient full20-token contexts')
  for case in report['cases']:
   for arm in case['arms']:
    for prediction in arm['result']['predictions']:require(sha(Path(prediction['path']))==prediction['sha256'],'Recorded logits changed')
  require(all(sha(Path(p))==h for p,h in report['inputs'].items()),'Frozen inputs changed')
  report.update(completed=True,passed=True,guard_after=guard(BASE/'../benchmark-status'))
 except BaseException as e:report['error']=dict(type=type(e).__name__,message=str(e));raise
 finally:
  if active is not None:active.close()
  save(out,report)
 print(out)
if __name__=='__main__':main()
