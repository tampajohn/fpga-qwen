#!/usr/bin/env python3
"""Root-run extra common-prefix PV edges versus tested query-row time32."""
from pathlib import Path
import hashlib,json,subprocess,sys
ROOT=Path(__file__).resolve().parents[2];B=ROOT/'build/native-cpu'
def sha(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def read(p):return json.loads(Path(p).read_text())
def guarded(name,args):subprocess.run([sys.executable,str(ROOT/'tools/benchmark_guard.py'),'--name',name,'--',sys.executable,*map(str,args)],cwd=ROOT,check=True)
reference=B/'prefill-query-row-pair-build/qwen_cpu';candidate=B/'prefill-query-row-pair-time128-build/qwen_cpu'
source=B/'extended-tutorial-fixtures/context3968.json';teacher=B/'extended-tutorial-fixtures/context3968-continuation1.json'
folder=B/'prefill-query-row-pair-time128-edge-fixtures';assert not folder.exists()
output=B/'prefill-query-row-pair-time128-explicit-edge-exact-gate.json';assert not output.exists()
paths=[source,teacher,Path(__file__).resolve()]+[ROOT/'tools'/n for n in ('capture_native_cpu_logits.py','compare_native_cpu_captures.py','native_cpu_chat.py','benchmark_guard.py')]
for binary in (reference,candidate):
 m=read(binary.parent/'manifest.json');assert m['completed'] and m['binary_sha256']==sha(binary)
 paths.extend((binary,binary.parent/'manifest.json'))
 for path,digest in m['source_sha256'].items():assert sha(path)==digest;paths.append(Path(path))
for n in (127,128,129,130):
 for arm in ('reference','candidate'):assert not (B/f'time128-edge-{n}-{arm}-logits').exists()
folder.mkdir();ids=read(source)
for n in (127,128,129,130):
 p=folder/f'prompt-{n}.json';p.write_text(json.dumps(ids[:n])+'\n');paths.append(p)
frozen={str(p):sha(p) for p in paths}
(folder/'manifest.json').write_text(json.dumps(dict(source=str(source),source_sha256=sha(source),lengths=[127,128,129,130],note='Fixed leading IDs from existing3968fixture.130added because paired rows atfreshposition0 have odd common causal length;130exercises common129 across128boundary.',inputs=frozen),indent=2)+'\n')
refs=[];cands=[]
for n in (127,128,129,130):
 for arm,binary,collection in [('reference',reference,refs),('candidate',candidate,cands)]:
  assert all(sha(p)==h for p,h in frozen.items())
  directory=B/f'time128-edge-{n}-{arm}-logits'
  guarded(f'Time128edge{n} {arm}',[ROOT/'tools/capture_native_cpu_logits.py','--binary',binary,'--weights',B/'weights.bin','--prompt',folder/f'prompt-{n}.json','--continuation',teacher,'--tokens',2,'--threads',12,'--prefill-threads',6,'--directory',directory]);collection.append(directory/'report.json')
guarded('Time128explicit edge relative identity',[ROOT/'tools/compare_native_cpu_captures.py','--reference',*refs,'--candidate',*cands,'--output',output])
j=read(output);assert j['completed'] and j['passed'] and j['compared_arrays']==16
assert all(sha(p)==h for p,h in frozen.items())
print('16explicit edge arrays matched; relative identity only, not independent absolute NumPy qualification.')
