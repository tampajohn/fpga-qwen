#!/usr/bin/env python3
"""Root-only predeclared incremental comparison; no automatic release selection."""
from pathlib import Path
import json
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[2]
BASE = ROOT / 'build/native-cpu'
TAG = 'queryrow-b1-pv-time32'

def guarded(name, arguments):
    subprocess.run([sys.executable, str(ROOT / 'tools/benchmark_guard.py'),
                    '--name', name, '--', sys.executable, *map(str, arguments)],
                   cwd=ROOT, check=True)

for suite in ('standard', 'extended'):
    stem = BASE / f'{TAG}-vs-queryrow-{suite}-confirmation'
    assert not stem.with_suffix('.json').exists()
    assert not Path(str(stem) + '-logs').exists()
    assert not (BASE / f'{TAG}-vs-queryrow-{suite}-analysis.json').exists()

guarded('B1 PV complete relative prerequisites',
        [BASE / 'check-b1pv-relative-prerequisites.py', 'final'])
for suite, tokens, rounds in [('standard', 32, 20), ('extended', 128, 10)]:
    output = BASE / f'{TAG}-vs-queryrow-{suite}-confirmation.json'
    guarded(f'B1 PV versus prefill-only {suite} {rounds} pairs',
            [ROOT / 'tools/benchmark_native_cpu.py',
             '--binary', BASE / f'{TAG}-build/qwen_cpu',
             '--reference-binary', BASE / 'prefill-query-row-pair-build/qwen_cpu',
             '--weights', BASE / 'weights.bin', '--threads', 12,
             '--prefill-threads', 6, '--mode', 'confirmation',
             '--rounds', rounds, '--warmups', 1, '--context-suite', suite,
             '--tokens', tokens, '--output', output])
    report = json.loads(output.read_text())
    assert report['completed'] and report['paired_sampling_complete']
    assert report['rounds'] == rounds and len(report['visits']) == 2 * rounds
    subprocess.run([sys.executable, str(ROOT / 'tools/analyze_native_phase0.py'),
                    '--report', str(output),
                    '--baseline', 'native-reference-prefill-query-row-pair-build',
                    '--quality-gate', str(BASE / f'{TAG}-{"extended-quality" if suite == "extended" else "quality"}.json'),
                    '--output', str(BASE / f'{TAG}-vs-queryrow-{suite}-analysis.json')],
                   cwd=ROOT, check=True)
print('Incremental B1 comparison completed; independent numerical failures remain documented.')
